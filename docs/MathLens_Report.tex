% ABSTRACT
% ==========================================
\begin{abstract}
\noindent \textbf{MathLens} is an interactive, web-based software laboratory designed to bridge abstract mathematical concepts in linear algebra with concrete spatial visual representations in digital image processing. Digital images are fundamentally formatted as two-dimensional numerical arrays (grayscale intensity matrices) or three-dimensional tensor structures (RGB channels). MathLens enables students, researchers, and educators to manipulate raw matrix elements, execute scalar multiplications ($c \cdot A$), decompose multi-channel color tensors, and perform linear coordinate transformations ($X' = A X$) with instant 60fps canvas visual feedback. This report documents the complete architectural design, mathematical foundations, component system, UI/UX aesthetics, and implementation details of the MathLens project.
\end{abstract}

\section{Introduction}
% ==========================================

\subsection{Problem Motivation}
Linear algebra is a core cornerstone of modern computer science, software engineering, robotics, data science, and artificial intelligence. However, students frequently face cognitive hurdles when transitioning from scalar arithmetic to matrix operations and vector space transformations. Standard textbook presentations show matrices as static rectangular grids of numbers, lacking interactive or visual feedback.

\subsection{Utility of Visual Mathematical Exploration}
Digital image processing provides a intuitive domain for demonstrating linear algebra:
\begin{itemize}
    \item \textbf{Direct Mapping}: Every pixel on a monitor corresponds to an array element $A_{i,j} \in [0, 255]$.
    \item \textbf{Algebraic Operations as Visual Filters}: Matrix addition corresponds to image blending, while scalar multiplication ($c \cdot A$) corresponds to brightness scaling.
    \item \textbf{Spatial Linear Transformations}: A $2 \times 2$ transformation matrix physically rotates, scales, shears, or reflects pixel coordinate vectors on a 2D Cartesian plane.
\end{itemize}

\subsection{Project Objective}
The primary goal of the \textbf{MathLens} project—developed by Mayank Soni, Naman Kumar, and Krishana Yadav under the guidance of Prof. Payal Wadhwa—is to engineer an accessible, web-native linear algebra laboratory that provides instant visual feedback, frame-accurate step animations, and intuitive interactive matrix controls.

\section{Problem Statement and System Overview}
% ==========================================

\subsection{Problem Statement}
To construct an interactive, client-side web application that connects matrix mathematics to digital image processing without requiring server-side compute delays or binary plugin dependencies.

\subsection{System Overview}
MathLens connects algebraic inputs directly to HTML5 Canvas rendering contexts. When a user adjusts an element in an interactive matrix grid, moves a scalar multiplier slider, or selects a coordinate transformation matrix, the system processes the underlying matrix calculations through dedicated mathematical engine modules and renders the result live.

\section{Implemented Mathematical Formulations}
% ==========================================

\subsection{Grayscale Pixel Matrix Representation}
An $M \times N$ grayscale image is represented as a matrix $A \in \mathbb{R}^{M \times N}$:
\begin{equation}
A = \begin{bmatrix}
a_{1,1} & a_{1,2} & \dots & a_{1,N} \\
a_{2,1} & a_{2,2} & \dots & a_{2,N} \\
\vdots & \vdots & \ddots & \vdots \\
a_{M,1} & a_{M,2} & \dots & a_{M,N}
\end{bmatrix}, \quad a_{i,j} \in [0, 255] \subset \mathbb{Z}
\end{equation}

\subsection{Scalar Matrix Multiplication}
Given a scalar constant $c \ge 0$, the multiplied matrix $B = c \cdot A$ is defined element-wise by:
\begin{equation}
B_{i,j} = \text{clamp}(c \cdot a_{i,j}), \quad \text{where } \text{clamp}(x) = \min(255, \max(0, \lfloor x \rceil))
\end{equation}

\subsection{RGB Tensor Channel Decomposition}
An RGB image is a 3D tensor $I \in \mathbb{R}^{M \times N \times 3}$:
\begin{equation}
I(i, j) = \begin{bmatrix} R_{i,j} \\ G_{i,j} \\ B_{i,j} \end{bmatrix}
\end{equation}

\subsection{2D Linear Coordinate Transformations}
For coordinate vector $\vec{x} = \begin{bmatrix} x \\ y \end{bmatrix}$ and transformation matrix $A = \begin{bmatrix} a & b \\ c & d \end{bmatrix}$:
\begin{equation}
\vec{x}' = A \vec{x} = \begin{bmatrix} a & b \\ c & d \end{bmatrix} \begin{bmatrix} x \\ y \end{bmatrix} = \begin{bmatrix} ax + by \\ cx + dy \end{bmatrix}
\end{equation}

\subsubsection{Rotation Matrix $R(\theta)$}
\begin{equation}
R(\theta) = \begin{bmatrix} \cos\theta & -\sin\theta \\ \sin\theta & \cos\theta \end{bmatrix}
\end{equation}

\subsubsection{Scaling Matrix $S(s_x, s_y)$}
\begin{equation}
S(s_x, s_y) = \begin{bmatrix} s_x & 0 \\ 0 & s_y \end{bmatrix}
\end{equation}

\subsubsection{Shearing Matrix $H(k_x)$}
\begin{equation}
H(k_x) = \begin{bmatrix} 1 & k_x \\ 0 & 1 \end{bmatrix}
\end{equation}

\subsubsection{Reflection Matrix $M_x$}
\begin{equation}
M_x = \begin{bmatrix} 1 & 0 \\ 0 & -1 \end{bmatrix}
\end{equation}

\section{Implementation Details}
% ==========================================

\subsection{Mathematical Engines}
\begin{itemize}
    \item \texttt{src/core/mathEngine.js}: Contains core operations for matrix allocation, clamping, and scalar multiplication.
    \item \texttt{src/core/transformEngine.js}: Performs $2 \times 2$ vector transformations and preset generation.
\end{itemize}

\subsection{Visualization Components}
\begin{itemize}
    \item \texttt{PixelCanvas.jsx}: Draws pixel arrays onto HTML5 Canvas with device-pixel-ratio scaling.
    \item \texttt{MatrixGrid.jsx}: Renders editable matrix brackets and input elements.
    \item \texttt{RGBExplodedView.jsx}: 4-column tabbed channel decomposition view with live cell editing.
    \item \texttt{CoordinateTransformCanvas.jsx}: Draws Cartesian axes, grid lines, and transformed vectors.
\end{itemize}

\section{UI/UX Design System}
% ==========================================

MathLens features a dual-theme system:
\begin{itemize}
    \item \textbf{Dark Mode}: Slate background (\texttt{\#0B0F17}) with muted steel-teal/blue accents (\texttt{\#0F766E} / \texttt{\#0369A1}).
    \item \textbf{Light Mode}: Soft cream background (\texttt{\#F3F6F0}) with smooth olive accents (\texttt{\#5C7249} / \texttt{\#465935}) and ambient dot-mesh texture overlays.
    \item \textbf{Typography}: Google Fonts \textit{Lora} (serif headings), \textit{Plus Jakarta Sans} (body), and \textit{JetBrains Mono} (matrix numbers).
\end{itemize}

\section{Current Limitations and Future Scope}
% ==========================================

\subsection{Current Limitations}
\begin{itemize}
    \item Matrix size is currently standardized to $4 \times 4$ for optimal classroom display.
    \item Transformations are limited to 2D planar matrices.
\end{itemize}

\subsection{Future Scope}
\begin{itemize}
    \item Arbitrary user image upload and real-time matrix conversion.
    \item 3D tensor visualizations for deep learning feature maps.
    \item Interactive Eigenvalue ($\lambda$) and Eigenvector ($\vec{v}$) visual decomposition ($A \vec{v} = \lambda \vec{v}$).
\end{itemize}
